if (!function_exists('e1_donate_campaign')) {
  /* Returns the due date (DateTimeImmutable) of the campaign running on $today, or null. */
  function e1_donate_campaign(string $anchor, int $months, int $days, DateTimeImmutable $today) {
    $tz   = $today->getTimezone();
    $a    = new DateTimeImmutable($anchor, $tz);
    $day  = (int)$a->format('j');
    $base = $a->modify('first day of this month');
    $diff = ((int)$today->format('Y') - (int)$a->format('Y')) * 12 + (int)$today->format('n') - (int)$a->format('n');
    $n    = intdiv($diff, $months) * $months;
    if ($diff < 0 && $diff % $months) $n -= $months;              // floor for dates before the anchor
    foreach ([$n, $n + $months] as $k) {                         // the due date in this cycle or the next
      $m   = $base->modify(($k >= 0 ? '+' : '') . $k . ' months');
      $due = $m->setDate((int)$m->format('Y'), (int)$m->format('n'), min($day, (int)$m->format('t')));
      $start = $due->modify('-' . ($days - 1) . ' days');
      $t = $today->setTime(0, 0);
      if ($t >= $start && $t <= $due) return $due;
    }
    return null;
  }
}

<script>
(function(){
  var KEY='e1DonateDismissed', card=document.getElementById('e1-donate');
  if(!card) return;
  var id=card.getAttribute('data-campaign'), preview=card.hasAttribute('data-preview');
  try{ if(!preview && localStorage.getItem(KEY)===id){ card.hidden=true; return; } }catch(e){}
  function remember(){ if(preview) return; try{ localStorage.setItem(KEY,id); }catch(e){} }
  card.querySelectorAll('[data-e1-dismiss]').forEach(function(el){
    el.addEventListener('click',function(){
      remember();
      if(el.tagName!=='A') card.hidden=true;
      else setTimeout(function(){ card.hidden=true; },300);   // Ko-fi opens in a new tab
    });
  });
})();
</script>
